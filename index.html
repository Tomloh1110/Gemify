async function fetchGeminiTasks(title, role) { // 拼接 Key const _part1 = "AIzaSyC1DSvs"; const _part2 = "fpU_xlUf4yYro"; const _part3 = "9UXcj_XgfR9s10"; const _fullKey = _part1 + _part2 + _part3; // 诊断点 1: 如果 gemini-3 报错，尝试换成 gemini-1.5-flash 看看是否是模型名问题 const MODEL_NAME = "gemini-3-flash-preview"; const API_URL = `https://generativelanguage.googleapis.com/v1beta/models/${MODEL_NAME}:generateContent?key=${_fullKey}`; const prompt = `You are the Gemify Evolution Engine. Break down "${title}" for a ${role} into a granular progression. Requirements: 1. Number of stages must be between 6 and 19. 2. XP MUST be multiples of 5 (5, 10, 15, 20, 25...). Start low and increase per stage. 3. Provide a resource_url from professional academies (W3Schools, MDN, Coursera, Udemy, official docs) or YouTube. 4. Suggest 3 essential Apps/Tools. Return ONLY a valid JSON object: { "tasks": [{"stage": 1, "name": "...", "xp": 5, "url": "..."}], "apps": ["App1", "App2"] }. No markdown, no preamble.`; try { const response = await fetch(API_URL, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ contents: [{ parts: [{ text: prompt }] }] }) }); if (!response.ok) { const errorData = await response.json(); // 这里的弹窗非常重要，请告诉我它显示的错误内容！ alert(`[Gemini API Error] Status: ${response.status}\nMessage: ${errorData.error.message}`); throw new Error("API_RESPONSE_NOT_OK"); } const data = await response.json(); let text = data.candidates[0].content.parts[0].text; // 诊断点 2: 更加鲁棒的 JSON 提取逻辑 // 有时 AI 会返回 "Here is the JSON: { ... }"，我们需要提取 {} 之间的内容 const jsonMatch = text.match(/\{[\s\S]*\}/); if (jsonMatch) { text = jsonMatch[0]; } window.lastEngine = "GEMINI_3_FLASH_PREVIEW_ACTIVE"; return JSON.parse(text); } catch (e) { console.error("AI Error:", e); // 如果是由于模型名不存在导致的 404，这里会触发 window.lastEngine = "FALLBACK_PROTOCOL"; return null; } }
